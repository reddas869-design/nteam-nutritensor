cmake_minimum_required(VERSION 3.24)
project(nteam_normalization_lab LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(NTEAM_WITH_CUDA "Build the optional GPU profiling exercise" ON)
enable_testing()
add_executable(normalize_reference_test src/reference_test.cpp)
target_include_directories(normalize_reference_test PRIVATE include)
add_test(NAME normalize_reference COMMAND normalize_reference_test)
if(NTEAM_WITH_CUDA)
    set(CMAKE_CUDA_ARCHITECTURES 86 CACHE STRING "Target GPU compute capability; RTX 3060 is 86")
    enable_language(CUDA)
    add_executable(normalize_benchmark src/normalize_benchmark.cu)
    target_include_directories(normalize_benchmark PRIVATE include)
    set_target_properties(normalize_benchmark PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    target_compile_options(normalize_benchmark PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
endif()
